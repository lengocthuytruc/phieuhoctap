<script>
    // URL Web App Google Apps Script cố định mới nhất của cô Lê Ngọc Thủy Trúc:
    const SCRIPT_URL = "https://script.google.com/macros/s/AKfycbxQ41XgYSOMuD-sewN6r3gryCSvBUCpYVu-Xmao-4u7onJN8P1KMaCrgWAwzk-EU0HV/exec";
    const LESSON_PREFIX = "PHT_TINHOC_BAI6_";

    // Khóa định danh duy nhất theo Lớp + Họ tên
    function getStudentKey(className, name) {
        const cleanClass = className.toUpperCase().replace(/\s+/g, "");
        const cleanName = name.toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "").replace(/\s+/g, "_");
        return `${LESSON_PREFIX}${cleanClass}_${cleanName}`;
    }

    function lockForm() {
        document.querySelectorAll("input, select, button").forEach(el => el.disabled = true);
    }

    function submitQuiz() {
        const name = document.getElementById("studentName").value.trim();
        const className = document.getElementById("studentClass").value.trim().toUpperCase();

        if (!name || !className) {
            alert("Em vui lòng nhập đầy đủ Họ tên và Lớp trước khi nộp bài nhé!");
            return;
        }

        // Lấy đáp án học sinh chọn
        const q1 = document.querySelector('input[name="q1"]:checked')?.value;
        const q2 = document.querySelector('input[name="q2"]:checked')?.value;
        const q3a = document.querySelector('input[name="q3a"]:checked')?.value;
        const q3b = document.querySelector('input[name="q3b"]:checked')?.value;
        const q3c = document.querySelector('input[name="q3c"]:checked')?.value;
        const q3d = document.querySelector('input[name="q3d"]:checked')?.value;
        const q4_1 = document.getElementById("q4_1").value;
        const q4_2 = document.getElementById("q4_2").value;
        const q4_3 = document.getElementById("q4_3").value;
        const q4_4 = document.getElementById("q4_4").value;
        const q5 = document.querySelector('input[name="q5"]:checked')?.value;

        if (!q1 || !q2 || !q3a || !q3b || !q3c || !q3d || !q4_1 || !q4_2 || !q4_3 || !q4_4 || !q5) {
            alert("Em hãy hoàn thành tất cả các câu hỏi từ Câu 1 đến Câu 5 trước khi bấm Nộp bài nhé!");
            return;
        }

        // Chấm điểm
        let score = 0;

        if (q1 === "B") {
            score += 2;
            showFeedback("fb-q1", true, "Đúng (+2 điểm)");
        } else {
            showFeedback("fb-q1", false, "Chưa đúng. Đáp án đúng: Giúp văn bản dễ đọc và được trình bày đẹp hơn.");
        }

        if (q2 === "C") {
            score += 2;
            showFeedback("fb-q2", true, "Đúng (+2 điểm)");
        } else {
            showFeedback("fb-q2", false, "Chưa đúng. Đáp án đúng: Cỡ chữ.");
        }

        let s3 = 0;
        if (q3a === "D") s3 += 0.5;
        if (q3b === "S") s3 += 0.5;
        if (q3c === "D") s3 += 0.5;
        if (q3d === "S") s3 += 0.5;
        score += s3;
        showFeedback("fb-q3", s3 === 2, `Đạt ${s3}/2 điểm (Đáp án chuẩn: a-Đúng, b-Sai, c-Đúng, d-Sai)`);

        let s4 = 0;
        if (q4_1 === "d") s4 += 0.5;
        if (q4_2 === "b") s4 += 0.5;
        if (q4_3 === "a") s4 += 0.5;
        if (q4_4 === "c") s4 += 0.5;
        score += s4;
        showFeedback("fb-q4", s4 === 2, `Đạt ${s4}/2 điểm (Đáp án chuẩn: 1-d, 2-b, 3-a, 4-c)`);

        if (q5 === "D") {
            score += 2;
            showFeedback("fb-q5", true, "Đúng (+2 điểm)");
        } else {
            showFeedback("fb-q5", false, "Chưa đúng. Đáp án đúng: Bố trí hình ảnh.");
        }

        const studentKey = getStudentKey(className, name);
        const existingData = localStorage.getItem(studentKey);
        const isRepeat = Boolean(existingData);

        if (!isRepeat) {
            localStorage.setItem(studentKey, JSON.stringify({ name, className, score }));
        }

        lockForm();

        // Hiển thị kết quả trên màn hình
        const panel = document.getElementById("resultPanel");
        panel.style.display = "block";
        if (isRepeat) {
            const firstScore = JSON.parse(existingData).score;
            panel.style.background = "#fefce8";
            panel.style.borderColor = "#facc15";
            document.getElementById("resultTitle").style.color = "#b45309";
            document.getElementById("resultTitle").innerText = "CẢNH BÁO: EM ĐÃ NỘP BÀI NÀY TRƯỚC ĐÓ!";
            document.getElementById("scoreDisplay").innerText = `Lần 1: ${firstScore}đ | Lần này: ${score}đ`;
        } else {
            document.getElementById("scoreDisplay").innerText = `${score} / 10 điểm`;
        }
        document.getElementById("studentSummary").innerText = `Học sinh: ${name} - Lớp: ${className}`;
        panel.scrollIntoView({ behavior: "smooth" });

        // Gửi dữ liệu về Google Sheets (nếu nộp lần 2 trở đi, Apps Script sẽ tự tô màu vàng & ghi chú NỘP LẦN 2)
        const payload = {
            lesson: "Bài 6: Định dạng kí tự và bố trí hình ảnh trong văn bản",
            name: name,
            className: className,
            score: score,
            detail: `C1:${q1} | C2:${q2} | C3:${q3a},${q3b},${q3c},${q3d} | C4:1-${q4_1},2-${q4_2},3-${q4_3},4-${q4_4} | C5:${q5}`
        };

        fetch(SCRIPT_URL, {
            method: "POST",
            mode: "no-cors",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify(payload)
        })
        .then(() => {
            if (isRepeat) {
                document.getElementById("syncStatus").innerText = "⚠ Hệ thống đã ghi nhận lượt nộp lại này vào bảng điểm của cô Lê Ngọc Thủy Trúc!";
            } else {
                document.getElementById("syncStatus").innerText = "✔ Đã gửi kết quả thành công về bảng điểm của cô Lê Ngọc Thủy Trúc!";
            }
        })
        .catch(() => {
            document.getElementById("syncStatus").innerText = "⚠ Lỗi kết nối mạng, em hãy chụp màn hình kết quả này gửi cô nhé!";
        });
    }
